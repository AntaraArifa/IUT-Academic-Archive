    // ---------- TC‑0009 ----------
    [Fact]
    public void TC0009_FetchArrivalTime_Is_SchedulePlusDuration()
    {
        
        var scheduler = new Flight();
        scheduler.FlightScheduler();              
        Flight flight = scheduler.GetFlightList().First();

        
        DateTime departure = DateTime.ParseExact(
            flight.GetFlightSchedule(),            
            "dddd, dd MMMM yyyy, HH:mm tt ",
            CultureInfo.InvariantCulture);

       
        var parts = flight.GetFlightTime().Split(':');
        int hours = int.Parse(parts[0]);
        int minutes = int.Parse(parts[1]);
        DateTime expectedArrival = departure.AddHours(hours)
                                            .AddMinutes(minutes);

        // Act
        string actualArrivalText = flight.FetchArrivalTime();  
        DateTime actualArrival = DateTime.ParseExact(
            actualArrivalText,
            "ddd, dd-MM-yyyy HH:mm tt",
            CultureInfo.InvariantCulture);

        // Assert
        Assert.Equal(expectedArrival, actualArrival);
    }

    {
        var scheduler = new Flight();
        scheduler.FlightScheduler();
        var flight = scheduler.GetFlightList().First();

        var cust = new Customer("Cancel", "cancel@ex.com", "pw", "0176", "Dhaka", 31);
        Customer.CustomerCollection.Add(cust);
        var res = new FlightReservation();
        res.BookFlight(flight.GetFlightNumber(), 6, cust.UserID);

        int seatsAfterBooking = flight.GetNoOfSeats();

        var originalIn = Console.In;
        Console.SetIn(new StringReader(input));
        res.CancelFlight(cust.UserID);
        Console.SetIn(originalIn);
